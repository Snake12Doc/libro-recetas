% --- DEFINICIÓN DEL ESTILO DE LA RECETA CORREGIDO ---
\newtcolorbox{receta}[3][]{
    enhanced,
    colback=white,
    colframe=orange!80!black,
    fonttitle=\bfseries\Large,
    title={\faUtensils\ #2 \hfill \faClock\ #3},
    % En lugar de 'no hyperref', usamos esto para evitar el conflicto
    hypertarget={receta#2}, 
    attach boxed title to top center={yshift=-3mm},
    boxed title style={colback=orange!80!black, sharp corners},
    sharp corners,
    drop shadow,
    % Quitamos el 'code' que añade al índice desde aquí 
    before skip=12pt,
    after skip=0pt,
    #1
}

% --- COMANDO PARA EL ÍNDICE Y REFERENCIAS SIN WARNINGS ---
\newcommand{\nuevaReceta}[3]{% 
    \phantomsection % Crea el ancla para el índice y hyperref
    \addcontentsline{toc}{section}{#1}% Registra en el índice
    \begin{receta}{#1}{#2}% Abre la caja (Título y Tiempo)
    \label{#3}% Establece la etiqueta para usar con \pageref
}

% --- RECETAS CON IMAGEN ADAPTABLE ---
\newsavebox{\cuerpoReceta}
\newlength{\alturaImagenReceta}

\NewDocumentEnvironment{recetaConImagen}{m m m m m +b}{%
    \phantomsection
    \addcontentsline{toc}{section}{#1}%
    \begin{receta}[height fill]{#1}{#2}%
    \label{#3}%
    % Medimos primero el texto a su ancho definitivo.
    \setbox\cuerpoReceta=\vbox{%
        \hsize=\linewidth
        \linewidth=\hsize
        #6%
    }%
    \setlength{\alturaImagenReceta}{%
        \dimexpr\tcbtextheight-\ht\cuerpoReceta-\dp\cuerpoReceta\relax
    }%
    \ifdim\alturaImagenReceta<3cm
        \setlength{\alturaImagenReceta}{3cm}%
    \fi
    % La foto crece dentro del hueco calculado, siempre completa y proporcionada.
    \vbox to \alturaImagenReceta{%
        \vfil
        \hbox to \linewidth{%
            \hfil
            \includegraphics[
                width=0.95\linewidth,
                height=\dimexpr\alturaImagenReceta-3\baselineskip\relax,
                keepaspectratio
            ]{#4}%
            \hfil
        }%
        \vspace{0.3em}%
        \parbox{\linewidth}{\centering\small\itshape #5}%
        \vfil
    }%
    \nointerlineskip
    \usebox{\cuerpoReceta}%
    \end{receta}%
}{}
