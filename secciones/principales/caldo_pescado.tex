\nuevaReceta{Pescado con Caldo y Patatas}{4 personas}{50 min}{receta:caldo_pescado}

    \origenReceta{\faHome\ De casa · Receta de familia}

    \begin{minipage}[t]{0.35\textwidth}
        \section*{Ingredientes}
        \begin{itemize}[leftmargin=*]
            \item 800 g de pescado blanco limpio, en lomos o trozos
            \item 700 g de patatas
            \item 1 cebolla
            \item 1 tomate maduro
            \item 2 dientes de ajo
            \item 1 hoja de laurel
            \item perejil fresco
            \item aceite de oliva
            \item 1,2 litros de agua o caldo de pescado, aprox.
            \item sal
        \end{itemize}
    \end{minipage}
    \hfill
    \begin{minipage}[t]{0.6\textwidth}
        \section*{Preparación}
        \begin{enumerate}
            \item \textbf{Preparar:} Pelar las patatas y chascarlas en trozos medianos. Picar la cebolla, el tomate y los ajos.
            \item \textbf{Sofrito:} Calentar un poco de aceite en una olla y rehogar la cebolla y los ajos. Añadir el tomate y cocinar durante 5-8 minutos, hasta que reduzca.
            \item \textbf{Cocer las patatas:} Incorporar las patatas, el laurel y el perejil. Cubrir con el agua o caldo, salar y cocer a fuego medio durante 18-20 minutos, hasta que las patatas estén casi tiernas.
            \item \textbf{Añadir el pescado:} Salar ligeramente los trozos de pescado e incorporarlos a la olla. Cocer a fuego suave durante 8-10 minutos, según su grosor.
            \item \textbf{Reposar y servir:} Apagar el fuego, dejar reposar 5 minutos y servir el pescado con las patatas y su caldo bien caliente.
        \end{enumerate}
    \end{minipage}

    \vspace{1em}
    \begin{tcolorbox}[colback=orange!5, colframe=orange!50, title=Consejo, size=small]
        Añade el pescado cuando las patatas estén casi hechas: así quedará jugoso y no se deshará dentro del caldo.
    \end{tcolorbox}
\end{receta}
