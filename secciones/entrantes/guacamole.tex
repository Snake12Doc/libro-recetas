\nuevaReceta{Guacamole Casero}{4 personas}{15 min}{receta:guacamole}
    
    \begin{minipage}[t]{0.35\textwidth}
        \section*{Ingredientes}
        \begin{itemize}[leftmargin=*]
            \item 3 aguacates maduros
            \item 1/2 cebolla morada
            \item 1 tomate pequeño
            \item cilantro fresco
            \item 1 lima y sal
        \end{itemize}
    \end{minipage}
    \hfill
    \begin{minipage}[t]{0.6\textwidth}
        \section*{Preparación}
        \begin{enumerate}
            \item Picar la cebolla y el tomate en trozos muy pequeños.
            \item En un bol, machacar la pulpa del aguacate con un tenedor.
            \item Mezclar todo y añadir el zumo de lima.
            \item Salar al gusto y decorar con cilantro.
        \end{enumerate}
    \end{minipage}

    \vspace{1em}
    \begin{tcolorbox}[colback=orange!5, colframe=orange!50, title=Truco del Chef, size=small]
        Ajusta la lima justo antes de servir. Si lo preparas con antelación, alisa la superficie y cúbrela con film a ras de la superficie para reducir al máximo el contacto con el aire.
    \end{tcolorbox}
\end{receta}
